interface PaginatedResponse<T> {
  items: T[];
  total: number;
  page: number;
  page_size: number;
  pages: number;
}

// 统计
interface StatsData { ... }
// 流水线状态
interface PipelineStatus { ... }
// 报道
interface Report { ... }
